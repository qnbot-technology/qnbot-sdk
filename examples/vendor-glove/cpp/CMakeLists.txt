cmake_minimum_required(VERSION 3.16)
project(qnbot_vendor_glove_example LANGUAGES CXX)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(qnbot_core CONFIG REQUIRED)
find_package(Threads REQUIRED)
add_executable(vendor_glove vendor_glove.cpp)
target_compile_features(vendor_glove PRIVATE cxx_std_17)
target_link_libraries(vendor_glove PRIVATE qnbot::core Threads::Threads)
if(CMAKE_CXX_COMPILER_ID MATCHES "Clang|GNU")
  target_compile_options(vendor_glove PRIVATE -Wall -Wextra -Werror -Wpedantic)
endif()
